public class Cita {
    private String fecha;
    private String hora;
    private String motivo;
    private String nombreMascota;

    public Cita(String fecha, String hora, String motivo, String nombreMascota) {
        this.fecha = fecha;
        this.hora = hora;
        this.motivo = motivo;
        this.nombreMascota = nombreMascota;
    }
}
